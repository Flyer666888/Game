"""Capture settled runtime frames and measure actual skeletal animation ticking."""
import json, os, time, unreal
_preview_start = time.monotonic()
_preview_samples = []
_preview_capture_count = 0
_preview_handle = None

def _preview_tick(delta):
    global _preview_capture_count
    elapsed = time.monotonic() - _preview_start
    target = 30.0 if _preview_capture_count == 0 else 32.0
    if elapsed < target:
        return
    world = unreal.find_object(None, '/Game/ReversedScale/Maps/L_CombatLabLite.L_CombatLabLite')
    if not world:
        return
    rows = []
    for actor in unreal.GameplayStatics.get_all_actors_of_class(world, unreal.SkeletalMeshActor):
        comp = actor.get_component_by_class(unreal.SkeletalMeshComponent)
        v = comp.get_socket_location('head')
        rows.append({'name': actor.get_name(), 'head_world': [v.x, v.y, v.z], 'animation_time': comp.get_position()})
    pawn = unreal.GameplayStatics.get_player_pawn(world, 0)
    _preview_samples.append({'game_time': unreal.GameplayStatics.get_time_seconds(world),
                             'pawn_class': pawn.get_class().get_path_name() if pawn else None, 'actors': rows})
    unreal.SystemLibrary.execute_console_command(world, 'HighResShot 1280x720')
    _preview_capture_count += 1
    unreal.log('[SettledPreview] capture ' + str(_preview_capture_count))
    if _preview_capture_count >= 2:
        out = os.path.join(unreal.Paths.project_dir(), 'Docs', 'AI', 'DesktopRuntimeAnimation.json')
        with open(out, 'w', encoding='utf-8') as f:
            json.dump(_preview_samples, f, indent=2)
        unreal.unregister_slate_post_tick_callback(_preview_handle)

_preview_handle = unreal.register_slate_post_tick_callback(_preview_tick)
unreal.log('[SettledPreview] delayed runtime capture registered')
